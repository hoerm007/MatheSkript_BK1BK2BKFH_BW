\adjustbox{valign=t}{\begin{minipage}{\potenzfunktionenLength}
    Funktionen vom Typ
    \[f(x)=a\cdot x^n, \quad a\neq 0, \quad n\in\N\]
    bezeichnen wir als Potenzfunktionen.

    Der Koeffizient \(a\) ist der Streckfaktor, wie wir ihn bereits von quadratischen Funktionen kennen.

    Die Hochzahl bzw. der Exponent \(n\) ist eine natürliche Zahl: \(\N=\{1,\,2,\,3,\,4,\,\dots\}\)

    Die Schaubilder der Potenzfunktionen teilen sich in drei verschiedene Formen auf:

\begin{Exercise}[title={Skizziere das Schaubild, gib die Symmetrie sowie das Verhalten für sehr große/kleine \(x\) an.}, label=potenzA1]

	\begin{minipage}{\textwidth}
		\begin{minipage}{0.5\textwidth}
			\begin{enumerate}[label=\alph*)]
				\item \(f(x)=-x^2\)
				\item \(g(x)=0,5x^3\)
				\item \(h(x)=2x^6\)
				\item \(i(x)=-\frac{3}{2}x^5\)
			\end{enumerate}
		\end{minipage}%
		\begin{minipage}{0.5\textwidth}
			\begin{enumerate}[label=\alph*)]
				\setcounter{enumi}{4}
				\item \(j(x)=0,1x^4\)
				\item \(k(x)=-\frac{3}{5}x^7\)
				\item \(l(x)=-\sqrt{2}x^4\)
				\item \(m(x)=3x^5\)
			\end{enumerate}
		\end{minipage}%
	\end{minipage}%
\end{Exercise}

\begin{Exercise}[title={Ordne die passenden Aussagen den Funktionen zu. Jeder Funktion können mehrere Aussagen zugeordnet werden. Die Aussagen können mehrfach verwendet werden.}, label=potenzA2]
	
	\begin{minipage}{\textwidth}
		\adjustbox{valign=t}{\begin{minipage}{0.5\textwidth}
			\begin{enumerate}[label=\alph*)]
				\item \(f(x)=-2x^4\)
				\item \(g(x)=\frac{1}{3}x^3\)
				\item \(h(x)=6x^6\)
				\item \(i(x)=-0,5x^5\)
			\end{enumerate}
		\end{minipage}}%
		\adjustbox{valign=t}{\begin{minipage}{0.5\textwidth}
			\begin{enumerate}[label=\arabic*)]
				\item Das Schaubild der Funktion ist punktsymmetrisch zum Ursprung.
				\item Es gilt \(x\to-\infty: f(x)\to\infty\)
				\item Das Schaubild ist parabelförmig.
				\item Der Streckfaktor ist negativ.
				\item Die Funktionswerte sind alle negativ oder Null.
				\item Der Exponent ist ungerade.
				\item Die Funktionswerte decken die reellen Zahlen ab. 
			\end{enumerate}
		\end{minipage}}%
	\end{minipage}%
\end{Exercise}

\begin{Answer}[ref=potenzA2]
	
	\adjustbox{valign=t}{\begin{minipage}{\textwidth}
			\begin{enumerate}[label=\arabic*)]
				\item Das Schaubild der Funktion ist punktsymmetrisch zum Ursprung: \(g(x)\) oder \(i(x)\)
				\item Es gilt \(x\to-\infty: f(x)\to\infty\): \(h(x)\) oder \(i(x)\)
				\item Das Schaubild ist parabelförmig: \(f(x)\) oder \(h(x)\)
				\item Der Streckfaktor ist negativ: \(f(x)\) oder \(i(x)\)
				\item Die Funktionswerte sind alle negativ oder Null: \(f(x)\)
				\item Der Exponent ist ungerade: \(g(x)\) oder \(i(x)\)
				\item Die Funktionswerte decken die reellen Zahlen ab: \(g(x)\) oder \(i(x)\)
			\end{enumerate}
	\end{minipage}}%
\end{Answer}
