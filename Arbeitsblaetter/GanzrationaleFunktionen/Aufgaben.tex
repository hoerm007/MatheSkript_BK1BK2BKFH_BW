\cohead{\Large\textbf{Vermischte Aufgaben}}
\fakesubsection{Vermischte Aufgaben}
Die folgenden Aufgaben sind Aufgaben aus alten Abschlussprüfungen nachempfunden. Es ist jeweils angegeben, wie viel Zeit in der Abschlussprüfung zur Bearbeitung der Aufgabe zur Verfügung stehen würde.
\begin{Exercise}[title={Zeitansatz 10 min, ohne Taschenrechner}, label=ganzFktVA_1]
	
	Gegeben ist eine Funktion \(f\) durch \(f(x)=-2(x+3)^2 x(x-1)^3,\ x\in\R\). Ihr Schaubild sei \(K_f\).
	\begin{enumerate}[label=\alph*)]
		\item Geben Sie das Verhalten für sehr große und sehr kleine \(x\) an und bestimmen Sie die Nullstellen.
		\item Skizzieren Sie \(K_f\) in ein geeignetes Koordinatensystem.
	\end{enumerate}
\end{Exercise}
\begin{Exercise}[title={Zeitansatz 5 min, ohne Taschenrechner}, label=ganzFktVA_2]

	\[x^4+\frac{1}{8}x=0\]
\end{Exercise}

\begin{Exercise}[title={Zeitansatz 12 min, mit Taschenrechner}, label=ganzFktVA_4]
	
	Eine Funktion \(g\) ist gegeben durch \(g(x)=x^4-5x^2+4,\ x\in\R\). Ihr Schaubild sei \(K_g\).
	
	\begin{enumerate}[label=\alph*)]
		\item Berechnen Sie die Nullstellen von \(g\). Nennen Sie \(g(x)\) in der Produktform.
		\item Zeichnen Sie \(K_g\) für \(-2\leq x\leq 2\) in ein geeignetes Koordinatensystem.
		\end{enumerate}	
\end{Exercise}
\begin{Exercise}[title={Zeitansatz 10 min, ohne Taschenrechner}, label=ganzFktVA_5]
	
	Eine Funktion \(h\) ist gegeben durch \(h(x)=0,4x^6-0,95x^3-1,35,\ x\in\R\). Ihr Schaubild sei \(K_g\).
	
	\begin{enumerate}[label=\alph*)]
		\item Berechnen Sie die Nullstellen und den \(y\)-Achsenabschnitt von \(g\).
		\item Zeichnen Sie \(K_g\) für \(-1,2\leq x\leq 1,7\) in ein geeignetes Koordinatensystem.
	\end{enumerate}	
\end{Exercise}

\begin{Answer}[ref=ganzFktVA_2]
	
	\begin{align*}
		x^4+\frac{1}{8}x&=0\\
		x\left(x^3+\frac{1}{8}\right)&=0\\
		\text{SvN: }x_1&=0\text{ oder}\\
		x^3+\frac{1}{8}&=0\ \Big\vert\ -\frac{1}{8}\\
		x^3&=-\frac{1}{8}\ \Big\vert\ \sqrt[\leftroot{0}\uproot{1}3]{\phantom{x}}\\
		x_2&=-\frac{1}{2}
	\end{align*}
\end{Answer}
\begin{Answer}[ref=ganzFktVA_3]
	\begin{enumerate}[label=\alph*)]
		\item Das Verhalten für \(x\to\pm\infty\) stimmt nicht, da der Grad 3 und der Leitkoeffizient 1 ist. Das Schaubild müsste also von links unten nach rechts oben verlaufen. Das gezeigte Schaubild verläuft aber von links oben nach rechts unten.
		\item Das Schaubild müsste bei \(x=0\) eine doppelte Nullstelle haben. Das gezeigte Schaubild hat dort eine einfache Nullstelle.
		\item Der \(y\)-Achsenabschnitt müsste bei \(i(0)=1\) liegt. Der \(y\)-Achsenabschnitt des gezeigten Schaubilds ist aber 0.
		\item Da alle Hochzahlen ungerade sind (3 und 1), müsste das Schaubild punktsymmetrisch zum Ursprung sein. Das gezeigte Schaubild ist nicht punktsymmetrisch zum Ursprung.
	\end{enumerate}
\end{Answer}
