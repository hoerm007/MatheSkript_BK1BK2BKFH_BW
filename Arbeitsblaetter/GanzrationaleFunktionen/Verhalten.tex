\begin{Exercise}[title={Gib das Verhalten für \(x\to \pm \infty \) an}, label=ganzVerA1]

	\begin{minipage}{\textwidth}
		\begin{minipage}{0.5\textwidth}
			\begin{enumerate}[label=\alph*)]
				\item \(f(x)=3x^3+2x-3\)
				\item \(g(x)=-2,5x^5+5x^3+2,5x^2\)
				\item \(h(x)=2x^6-3x^2-14x+1\)
				\item \(i(x)=-\frac{3}{5}x^5+2x^4+x^2-1\)
			\end{enumerate}
		\end{minipage}%
		\begin{minipage}{0.5\textwidth}
			\begin{enumerate}[label=\alph*)]
				\setcounter{enumi}{4}
				\item \(j(x)=-0,3x^6-x^4+2x^2-5,8\)
				\item \(k(x)=-\frac{7}{5}x^7+\frac{8}{7}x^6-\frac{11}{6}x^2-\frac{12}{5}x\)
				\item \(l(x)=x\left(-x^3+2x^2+5\right)\)
				\item \(m(x)=5x^2\left(x-1\right)^2\)
			\end{enumerate}
		\end{minipage}%
	\end{minipage}
\end{Exercise}

\begin{Exercise}[title={Gib jeweils ein Beispiel für eine ganzrationale Funktion an, die die folgenden Eigenschaften hat:}, label=ganzVerA2]
	
	\begin{minipage}{\textwidth}
		\adjustbox{valign=t}{\begin{minipage}{0.5\textwidth}
			\begin{enumerate}[label=\alph*)]
				\item \(x\to\infty\colon f(x)\to\infty\)
				\item \(x\to-\infty\colon g(x)\to-\infty\)
				\item \(x\to\infty\colon h(x)\to\infty\) und 
				
				\(x\to-\infty\colon h(x)\to\infty\)
				\item \(x\to\infty\colon i(x)\to-\infty\) und 
				
				\(x\to-\infty\colon i(x)\to\infty\)
				\item \(x\to\infty\colon j(x)\to\infty\) und das Schaubild soll punktsymmetrisch zum Ursprung sein.
			\end{enumerate}
		\end{minipage}}%
		\adjustbox{valign=t}{\begin{minipage}{0.5\textwidth}
			\begin{enumerate}[label=\alph*)]
				\setcounter{enumi}{5}
				\item \(x\to\infty\colon k(x)\to-\infty\) und der Grad soll 6 betragen.
				\item \(x\to\infty\colon l(x)\to\infty\), 
				
				\(x\to-\infty\colon l(x)\to-\infty\) und es soll sich um eine Potenzfunktion handeln. 
				\item \(x\to-\infty\colon m(x)\to-\infty\) und das Schaubild soll achsensymmetrisch zur \(y\)-Achse sein. 
			\end{enumerate}
		\end{minipage}}%
	\end{minipage}
\end{Exercise}

				\item Verhält sich wie \(3x^3\)

				\(x\to-\infty\colon\ f(x)\to-\infty\)

				\(x\to\infty\colon\ f(x)\to\infty\)
				\item Verhält sich wie \(-2,5x^5\)

				\(x\to-\infty\colon\ g(x)\to\infty\)

				\(x\to\infty\colon\ g(x)\to-\infty\)
				\item Verhält sich wie \(2x^6\)

				\(x\to-\infty\colon\ h(x)\to\infty\)

				\(x\to\infty\colon\ h(x)\to\infty\)
				\item Verhält sich wie \(-\frac{3}{5}x^5\)

				\(x\to-\infty\colon\ i(x)\to\infty\)

				\(x\to\infty\colon\ i(x)\to-\infty\)
			\end{enumerate}
		\end{minipage}%
		\begin{minipage}[t]{0.5\textwidth}
			\begin{enumerate}[label=\alph*)]
				\setcounter{enumi}{4}
				\item Verhält sich wie \(-0,3x^6\)

				\(x\to-\infty\colon\ j(x)\to-\infty\)

				\(x\to\infty\colon\ j(x)\to-\infty\)
				\item Verhält sich wie \(-\frac{7}{5}x^7\)

				\(x\to-\infty\colon\ k(x)\to\infty\)

				\(x\to\infty\colon\ k(x)\to-\infty\)

\begin{Answer}[ref=ganzVerA2]
	
	\begin{minipage}{\textwidth}
		\begin{minipage}[t]{0.5\textwidth}
			\begin{enumerate}[label=\alph*)]
				\item Bei geradem Grad muss der Leitkoeffizient positiv sein, z.B. \(f(x)=x^2\)
				
				Bei ungeradem Grad muss der Leitkoeffizient negativ sein, z.B. \(f(x)=-x^3\)
				\item Bei geradem Grad muss der Leitkoeffizient negativ sein, z.B. \(g(x)=-2x^4+x\)
				
				Bei ungeradem Grad muss der Leitkoeffizient positiv sein, z.B. \(g(x)=0,1x^3+5x^2-5\)
				\item Da das Schaubild in beide Richtungen nach oben verläuft, muss der Grad gerade sein und der Leitkoeffizient positiv, z.B. 
				
				\(h(x)=2x^4-x^3+x-8\)
				\item Da das Schaubild in verschiedene Richtungen verläuft, muss der Grad ungerade sein und da es nach rechts unten verläuft, muss der Leitkoeffizient negativ, z.B. \(i(x)=-2x^3-x^2+2x+1\)
			\end{enumerate}
		\end{minipage}%
		\begin{minipage}[t]{0.5\textwidth}
			\begin{enumerate}[label=\alph*)]
				\setcounter{enumi}{4}
				\item Da das Schaubild punktsymmetrische zum Ursprung sein soll, müssen alle Hochzahlen und damit auch der Grad ungerade sein. Da das Schaubild nach rechts oben verläuft, muss der Leitkoeffizient positiv sein, z.B. \(j(x)=x^3-2x\)
				\item Grad 6 und das Schaubild soll nach unten verlaufen, also muss der Leitkoeffizient negativ sein, z.B. \(k(x)=-x^6+2x^5-x^2+4\)
				\item Da das Schaubild in unterschiedliche Richtungen verläuft, muss der Grad ungerade sein. Da das Schaubild nach rechts oben verläuft, muss der Leitkoeffizient positiv sein. Da es sich um eine Potenzfunktion handelt, müssen alle Koeffizienten außer dem Leitkoeffizienten Null sein, z.B. 
				
				\(l(x)=2x^3\)
				\item Da das Schaubild achsensymmetrisch zur \(y\)-Achse sein soll, müssen alle Hochzahlen gerade oder Null sein. Da das Schaubild nach unten verläuft, muss der Leitkoeffizient negativ sein, z.B. \(m(x)=-2x^4+2x^2+3\)
			\end{enumerate}
		\end{minipage}%
	\end{minipage}
\end{Answer}
